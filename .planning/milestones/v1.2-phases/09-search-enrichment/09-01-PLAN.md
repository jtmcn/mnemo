   def get_chunk_range(self, book_id: str, start_seq: int, end_seq: int, limit: int = 20) -> list[Chunk]:
       start_seq = max(0, start_seq)
       rows = self.conn.execute(
           """SELECT * FROM chunks WHERE book_id = ? AND sequence BETWEEN ? AND ? ORDER BY sequence LIMIT ?""",

       ).fetchall()
