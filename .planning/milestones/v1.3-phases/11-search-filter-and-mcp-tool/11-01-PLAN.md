       def get_section_structure(self, book_id: str) -> list[list[str]]:
           rows = self.conn.execute(
               """
               SELECT section_path, MIN(sequence) as first_seq
               FROM chunks
               WHERE book_id = ? AND section_path != '[]'
               GROUP BY section_path
               ORDER BY first_seq
               """,
               (book_id,),
           ).fetchall()
